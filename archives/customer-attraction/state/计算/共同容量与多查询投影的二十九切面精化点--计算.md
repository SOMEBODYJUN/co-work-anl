\[
\psi=\min_{0\le s\le n-1,\ g\in\{0,1\}}
\left\{\frac{t-Mg}{p+t+s+g}
+\frac{a_ms+a_lg}{ell+1}+b_ms+b_lg
+\frac{\sum_q\mu_qq}{p+t+s+1}\right\},\quad M=\sum_q\mu_q.
\tag{P}

\[

\tag{BUDGET}
