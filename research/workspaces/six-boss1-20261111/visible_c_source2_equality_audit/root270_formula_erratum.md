\[
\begin{aligned}
\Phi_{30}(h;q,r,y,t)={}&
\frac{126000q}{b+1+r+y+t}

\frac{o}{3D}-\frac{y}{D+t}-\frac{t}{D+y}-\frac{y}{D(D+1)}

\end{aligned}
\]
